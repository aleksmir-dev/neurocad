🧠 Почему стандартный Django не подходит
Django изначально проектировался для статичных схем: модели пишутся в коде, миграции создаются и применяются. Runtime-создание моделей не поддерживается из коробки.

Однако с помощью динамического создания классов (type) можно создавать модели на лету, а схему БД — через обычные SQL-запросы (без миграций).

2. Сервис для генерации моделей
Используем type() для динамического создания классов:

python
# billing/dynamic/model_builder.py
from django.db import models
from django.contrib.contenttypes.models import ContentType
from .models import Entity, Field

class ModelBuilder:
    """Построитель динамических моделей"""

    def __init__(self, entity: Entity):
        self.entity = entity
        self._model = None

    def build(self):
        """Создаёт Django-модель на основе метаданных"""
        attrs = self._build_fields()
        attrs['__module__'] = 'billing.dynamic.models'  # обязательный атрибут
        attrs['Meta'] = self._build_meta()

        # Динамическое создание класса модели
        model_class = type(
            self.entity.name,           # имя класса
            (models.Model,),            # базовые классы
            attrs                       # атрибуты
        )

        self._model = model_class
        return model_class

    def _build_fields(self):
        """Создаёт поля модели"""
        fields = {}
        for field in self.entity.fields.all():
            field_kwargs = self._get_field_kwargs(field)
            fields[field.name] = self._get_field_class(field)(**field_kwargs)

        # Добавляем стандартные поля
        fields['created_at'] = models.DateTimeField(auto_now_add=True)
        fields['updated_at'] = models.DateTimeField(auto_now=True)
        fields['created_by'] = models.ForeignKey('users.User', on_delete=models.SET_NULL, null=True)

        return fields

    def _get_field_class(self, field: Field):
        """Возвращает класс поля Django по типу"""
        mapping = {
            'char': models.CharField,
            'text': models.TextField,
            'int': models.IntegerField,
            'float': models.FloatField,
            'bool': models.BooleanField,
            'date': models.DateField,
            'datetime': models.DateTimeField,
            'fk': models.ForeignKey,
        }
        return mapping.get(field.data_type, models.CharField)

    def _get_field_kwargs(self, field: Field):
        """Параметры для создания поля"""
        kwargs = {
            'verbose_name': field.verbose_name,
            'null': not field.is_required,
            'blank': not field.is_required,
        }

        if field.max_length and field.data_type == 'char':
            kwargs['max_length'] = field.max_length

        if field.data_type == 'fk' and field.target_entity:
            kwargs['to'] = field.target_entity.name  # имя целевой модели

        return kwargs

    def _build_meta(self):
        """Создаёт внутренний класс Meta"""
        class Meta:
            db_table = f"dynamic_{self.entity.name.lower()}"
            verbose_name = self.entity.name
            verbose_name_plural = f"{self.entity.name}ы"

        return Meta
3. Регистрация в ContentTypes для совместимости
Чтобы Django "знал" о динамических моделях, их нужно регистрировать в ContentTypes:

python
# billing/dynamic/registry.py
from django.contrib.contenttypes.models import ContentType
from .model_builder import ModelBuilder

class DynamicModelRegistry:
    """Реестр динамических моделей"""
    _models = {}

    @classmethod
    def get_model(cls, entity_id):
        """Получить или создать динамическую модель"""
        if entity_id not in cls._models:
            from .models import Entity
            entity = Entity.objects.get(id=entity_id)
            builder = ModelBuilder(entity)
            model = builder.build()

            # Регистрируем в ContentType
            ct, _ = ContentType.objects.get_or_create(
                app_label='dynamic',
                model=entity.name.lower()
            )
            cls._models[entity_id] = model

        return cls._models[entity_id]

# billing/reports/views.py
from django.http import JsonResponse
from django.db.models import Count, Sum, Avg
from billing.dynamic.registry import DynamicModelRegistry

def report_data(request, report_id):
    report = Report.objects.get(id=report_id)
    Model = DynamicModelRegistry.get_model(report.source_entity.id)

    queryset = Model.objects.all()

    # Применяем фильтры
    if report.filters:
        queryset = queryset.filter(**report.filters)

    # Группировка и агрегация
    if report.grouping:
        values = queryset.values(*report.grouping)
        aggregations = {}

        for field in report.fields:
            if field.get('aggregation') == 'count':
                aggregations[f"{field['name']}_count"] = Count(field['name'])
            elif field.get('aggregation') == 'sum':
                aggregations[f"{field['name']}_sum"] = Sum(field['name'])
            elif field.get('aggregation') == 'avg':
                aggregations[f"{field['name']}_avg"] = Avg(field['name'])

        if aggregations:
            queryset = values.annotate(**aggregations)

    data = list(queryset)
    return JsonResponse({'data': data, 'chart_config': report.chart_config})

⚠️ Критические предостережения
Проблема	Решение
Безопасность	Никогда не позволяйте пользователям выполнять произвольный SQL. Все операции — через Django ORM.
Производительность	Кэшируйте созданные модели (DynamicModelRegistry).
Резервное копирование	Динамические таблицы нужно бэкапить отдельно.
Миграции	Не используйте стандартные миграции. Управляйте схемой через SQL-запросы.
Индексы	Создавайте их вручную при добавлении полей.
📌 Итог: что вы получите
Пользователь создаёт сущность — через интерфейс вводит название, добавляет поля, выбирает типы, указывает связи.

Система создаёт таблицу — генерирует Django-модель, создаёт SQL-таблицу и индексы.

Пользователь строит отчёты — выбирает сущность, поля, группировку, тип графика.

Отчёт отображается — через ECharts с возможностью экспорта в Excel/PDF.

Вся эта система работает без перезагрузки сервера. Это сложно, но реализуемо. Готов помочь с реализацией любого блока.

// frontend/src/components/ReportBuilder.jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';
import FieldSelector from './FieldSelector';
import FilterBuilder from './FilterBuilder';
import DataPreview from './DataPreview';
import ChartPreview from './ChartPreview';

const ReportBuilder = () => {
    const [entities, setEntities] = useState([]);
    const [selectedEntity, setSelectedEntity] = useState(null);
    const [fields, setFields] = useState([]);
    const [filters, setFilters] = useState([]);
    const [grouping, setGrouping] = useState([]);
    const [chartType, setChartType] = useState('table');
    const [previewData, setPreviewData] = useState(null);
    const [loading, setLoading] = useState(false);
    const [reportId, setReportId] = useState(null);

    // Загрузка списка сущностей
    useEffect(() => {
        axios.get('/api/entities/').then(res => setEntities(res.data));
    }, []);

    // Обновление данных при изменении настроек
    useEffect(() => {
        if (selectedEntity) {
            refreshPreview();
        }
    }, [selectedEntity, fields, filters, grouping, chartType]);

    const refreshPreview = async () => {
        setLoading(true);
        try {
            const response = await axios.post('/api/reports/preview/', {
                source: { entity_id: selectedEntity.id },
                fields,
                filters,
                grouping,
                chart_type: chartType,
            });
            setPreviewData(response.data);
        } catch (error) {
            console.error('Ошибка загрузки данных:', error);
        } finally {
            setLoading(false);
        }
    };

    const saveReport = async () => {
        const response = await axios.post('/api/reports/', {
            name: 'Новый отчёт',
            source: { entity_id: selectedEntity.id },
            fields,
            filters,
            grouping,
            chart_type: chartType,
        });
        setReportId(response.data.id);
        alert('Отчёт сохранён');
    };

    return (
        <div className="report-builder">
            <div className="sidebar">
                <h3>Выбор данных</h3>
                <select onChange={e => setSelectedEntity(entities.find(e => e.id == e.target.value))}>
                    <option value="">Выберите сущность</option>
                    {entities.map(e => <option key={e.id} value={e.id}>{e.name}</option>)}
                </select>

                {selectedEntity && (
                    <>
                        <FieldSelector
                            entityId={selectedEntity.id}
                            fields={fields}
                            onChange={setFields}
                        />
                        <FilterBuilder
                            entityId={selectedEntity.id}
                            filters={filters}
                            onChange={setFilters}
                        />
                        <div className="grouping">
                            <h4>Группировка</h4>
                            <select multiple onChange={e => setGrouping([...e.target.selectedOptions].map(o => o.value))}>
                                {fields.map(f => (
                                    <option key={f.name} value={f.name}>{f.verbose}</option>
                                ))}
                            </select>
                        </div>
                        <div className="chart-type">
                            <h4>Тип визуализации</h4>
                            <select value={chartType} onChange={e => setChartType(e.target.value)}>
                                <option value="table">Таблица</option>
                                <option value="bar">Столбчатая диаграмма</option>
                                <option value="line">Линейный график</option>
                                <option value="pie">Круговая диаграмма</option>
                                <option value="area">Областная диаграмма</option>
                            </select>
                        </div>

                        <button onClick={saveReport} className="btn-save">Сохранить отчёт</button>
                    </>
                )}
            </div>

            <div className="preview">
                <div className="toolbar">
                    <button onClick={refreshPreview} disabled={loading}>Обновить</button>
                    {reportId && (
                        <>
                            <button onClick={() => window.open(`/api/reports/${reportId}/export/excel/`)}>📊 Excel</button>
                            <button onClick={() => window.open(`/api/reports/${reportId}/export/pdf/`)}>📄 PDF</button>
                        </>
                    )}
                </div>

                {loading && <div className="loader">Загрузка...</div>}

                {previewData && !loading && (
                    chartType === 'table'
                        ? <DataPreview data={previewData} />
                        : <ChartPreview type={chartType} data={previewData} config={previewData.chart_config} />
                )}
            </div>
        </div>
    );
};

export default ReportBuilder;

const DataPreview = ({ data }) => {

        headerName: col.verbose,
        width: col.width || 150,
        editable: true,  // включили inline-редактирование!
        sortable: true,
        filter: true,
        pinned: col.is_primary ? 'left' : null,  // фиксация ключевых полей

            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ [event.colDef.field]: event.newValue }),
        });
    };

    return (
        <div className="ag-theme-alpine" style={{ height: 500, width: '100%' }}>
            <AgGridReact
                rowData={data.rows}
                columnDefs={columns}
                pagination={true}
                paginationPageSize={25}
                onCellValueChanged={onCellValueChanged}
                enableSorting={true}
                enableFilter={true}
            />
        </div>
    );
};

    useEffect(() => {
        if (!chartRef.current) return;

        chart = echarts.init(chartRef.current);

        const option = {
            title: { text: data.title || 'Отчёт', left: 'center' },
            tooltip: { trigger: 'axis' },
            legend: { data: data.series.map(s => s.name) },
            xAxis: { type: 'category', data: data.categories },
            yAxis: { type: 'value' },
            series: data.series.map(s => ({
                name: s.name,
                type: type === 'bar' ? 'bar' : type === 'line' ? 'line' : 'line',
                data: s.data,
                areaStyle: type === 'area' ? {} : undefined,
                smooth: type === 'line',
            })),
        };

        chart.setOption(option);

        return () => chart?.dispose();
    }, [data, type]);

    return <div ref={chartRef} style={{ width: '100%', height: 500 }} />;
};

# billing/reports/api.py
from rest_framework.decorators import api_view
from rest_framework.response import Response
from django.db import connection
from billing.dynamic.registry import DynamicModelRegistry

@api_view(['POST'])
def preview_report(request):
    """Возвращает данные для предпросмотра отчёта"""
    config = request.data

    # Получаем динамическую модель
    entity_id = config['source']['entity_id']
    Model = DynamicModelRegistry.get_model(entity_id)

    # Строим запрос
    queryset = Model.objects.all()

    # Применяем фильтры
    for f in config.get('filters', []):
        lookup = f"{f['field']}__{f.get('op', 'exact')}"
        queryset = queryset.filter(**{lookup: f['value']})

    # Группировка и агрегация
    if config.get('grouping'):
        values = queryset.values(*config['grouping'])
        annotations = {}
        for field in config['fields']:
            if field.get('aggregation'):
                annotations[f"{field['name']}_{field['aggregation']}"] = {
                    'sum': models.Sum(field['name']),
                    'avg': models.Avg(field['name']),
                    'count': models.Count(field['name']),
                }[field['aggregation']]
        if annotations:
            queryset = values.annotate(**annotations)

    # Сбор данных для фронтенда
    rows = list(queryset)
    columns = [
        {'field': f['name'], 'verbose': f.get('verbose', f['name'])}
        for f in config['fields']
    ]

    # Подготовка для графика
    categories = []
    series_data = {}
    for row in rows:
        if config.get('grouping'):
            cat = '-'.join(str(row.get(g)) for g in config['grouping'])
            categories.append(cat)
            for field in config['fields']:
                if field.get('aggregation'):
                    series_data.setdefault(field['verbose'], []).append(row.get(f"{field['name']}_{field['aggregation']}", 0))

    return Response({
        'columns': columns,
        'rows': rows,
        'categories': categories,
        'series': [{'name': k, 'data': v} for k, v in series_data.items()],
        'chart_config': config.get('chart_config', {}),
    })

    # Заголовки

            if field.get('data_type') == 'bool':

    response = HttpResponse(content_type='application/vnd.ms-excel')

    wb.save(response)

📊 Что позволяет ECharts
ECharts — это не просто библиотека для графиков, а мощный инструмент визуализации данных. Он под капотом использует Canvas, что обеспечивает отличную производительность даже на больших данных.

В контексте вашего конструктора отчетов он позволяет реализовать:

Интерактивность: пользователь может наводить на элементы графиков, кликать по легенде, чтобы скрыть или показать ряды данных, и масштабировать области.

Разнообразие визуализаций: поддерживаются десятки типов диаграмм (столбчатые, линейные, круговые, точечные, тепловые карты, географические карты и многие другие), что идеально подходит для гибкой системы отчетов.

Адаптивность: графики автоматически подстраиваются под размер контейнера, в который они встроены.

Кастомизация: вы можете детально настроить внешний вид под корпоративный стиль (цвета, шрифты, легенды), что соответствует вашему требованию по дизайну.

class Report(models.Model):

    data = models.JSONField(default=dict, blank=True, verbose_name='Кэшированные данные')

    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        verbose_name = 'Отчёт'
        verbose_name_plural = 'Отчёты'

    def __str__(self):

            yAxis: { type: 'value' },
            series: [{ type: 'bar', data: data.values }]
        });
    }

    // Обработка событий блоков (удаление, настройки)
    function attachBlockEvents(widget) {
        widget.querySelector('.block-delete').addEventListener('click', () => grid.removeWidget(widget));
        widget.querySelector('.block-settings').addEventListener('click', () => openBlockSettings(widget));
    }

    // Окно настроек блока (WYSIWYG-редактирование)
    function openBlockSettings(widget) {
        const blockId = widget.getAttribute('gs-id');
        const blockType = widget.querySelector('.block-title').innerText.startsWith('Таблица') ? 'table' : 'chart';

        const modal = document.createElement('div');
        modal.className = 'modal';
        modal.innerHTML = `
            <div class="modal-content">
                <h3>Настройки блока</h3>
                <label>Заголовок: <input type="text" id="block-title" value="${widget.querySelector('.block-title').innerText.replace('Таблица: ', '').replace('График: ', '')}"></label>
                <label>Источник данных:
                    <select id="block-entity">
                        <option value="bill">Платежи</option>
                        <option value="sub">Подписки</option>
                        <option value="user">Пользователи</option>
                    </select>
                </label>
                <label>Поля (через запятую): <input type="text" id="block-fields" value="id,amount,status"></label>
                <button id="save-settings">Сохранить</button>
            </div>
        `;

        document.body.appendChild(modal);

        document.getElementById('save-settings').addEventListener('click', () => {
            const newTitle = document.getElementById('block-title').value;
            const newEntity = document.getElementById('block-entity').value;
            const newFields = document.getElementById('block-fields').value.split(',').map(f => f.trim());

            widget.querySelector('.block-title').innerText = `${blockType === 'table' ? 'Таблица' : 'График'}: ${newTitle}`;

            // Обновляем блок с новыми настройками
            if (blockType === 'table') {
                renderTable(blockId, { source: { entity: newEntity, fields: newFields } });
            } else {
                renderChart(blockId, { source: { entity: newEntity, field_x: 'created_at', field_y: 'amount' } });
            }

            modal.remove();
        });
    }

    // Сохранение отчёта
    async function saveReport() {
        const blocks = [];
        const widgets = grid.getGridItems();

        for (const widget of widgets) {
            const id = widget.getAttribute('gs-id');
            const blockType = widget.querySelector('.block-title').innerText.startsWith('Таблица') ? 'table' : 'chart';
            const title = widget.querySelector('.block-title').innerText.replace('Таблица: ', '').replace('График: ', '');

            blocks.push({
                id: id,
                type: blockType,
                title: title,
                position: {
                    x: parseInt(widget.getAttribute('gs-x')),
                    y: parseInt(widget.getAttribute('gs-y')),
                    w: parseInt(widget.getAttribute('gs-w')),
                    h: parseInt(widget.getAttribute('gs-h'))
                }
            });
        }

        const config = {
            layout: 'grid',
            columns: 12,
            blocks: blocks
        };

        await fetch(`/api/reports/${currentReportId}/`, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ config: config })
        });

        alert('Отчёт сохранён');
    }

    // Загрузка отчёта
    async function loadReport(reportId) {
        const response = await fetch(`/api/reports/${reportId}/`);
        const report = await response.json();
        currentReportId = reportId;

        // Очищаем сетку
        grid.removeAllWidgets();

        // Восстанавливаем блоки
        for (const block of report.config.blocks) {
            if (block.type === 'table') {
                addTableBlock(block);
            } else {
                addChartBlock(block);
            }
        }
    }

    // Инициализация
    document.addEventListener('DOMContentLoaded', () => {
        initGrid();

        document.getElementById('add-table').addEventListener('click', () => addTableBlock());
        document.getElementById('add-chart').addEventListener('click', () => addChartBlock());
        document.getElementById('save-report').addEventListener('click', saveReport);
        document.getElementById('report-list').addEventListener('change', (e) => {
            if (e.target.value) loadReport(e.target.value);
        });
    });
</script>

from rest_framework import viewsets, serializers
from rest_framework.decorators import action
from rest_framework.response import Response
from .models import Report
from billing.dynamic.registry import DynamicModelRegistry

class ReportSerializer(serializers.ModelSerializer):
    class Meta:
        model = Report
        fields = '__all__'

class ReportViewSet(viewsets.ModelViewSet):
    queryset = Report.objects.all()
    serializer_class = ReportSerializer

    @action(detail=True, methods=['post'])
    def render(self, request, pk=None):
        """Рендерит отчёт в HTML для просмотра (не редактирования)"""
        report = self.get_object()
        # Генерируем HTML на основе config
        return Response({'html': self._render_to_html(report.config)})

    @action(detail=False, methods=['post'])
    def data(self, request):
        """Возвращает данные для таблицы"""
        config = request.data
        Model = DynamicModelRegistry.get_model(config['source']['entity'])
        queryset = Model.objects.all()
        # ... фильтры
        rows = list(queryset.values(*config['source'].get('fields', [])))
        columns = [{'field': f, 'verbose': f} for f in config['source'].get('fields', [])]
        return Response({'rows': rows, 'columns': columns})

    @action(detail=False, methods=['post'])
    def chart_data(self, request):
        """Возвращает данные для графика"""
        config = request.data
        Model = DynamicModelRegistry.get_model(config['source']['entity'])
        # ... агрегация
        return Response({'categories': ['Янв', 'Фев', 'Мар'], 'values': [100, 200, 150]})
