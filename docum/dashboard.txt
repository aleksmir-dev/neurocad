Ещё ты мне предлагал ранее Dashboard. Дак вот я хочу, чтобы она никогда не пролистывалась вниз. Она должна состять из блоков nav - всегда фиксирована вверху, footer - всешда фиксирован внизу, left side-bar, right-side-bar - всегда фиксированы слева и справа и у них внутренняя прокрутка независима от всей страницы (т.к. страница никогда не прокручивается). В главной части страницы набор виджетов, коттрые каждый между собой независимы, можно двигать их границы, внутри них свои прокрутки, можно их добавлять динамически (виджеты), расставлять. Такое можно реализовать?
Да, это абсолютно реализуемо. Вы описываете архитектуру, которая в веб-разработке называется "фиксированный макет с независимыми прокрутками" (fixed layout with independent scroll areas). Это классическая компоновка современных десктопных приложений, перенесённая в веб.

🏗️ Архитектура макета

│           │   │                                     │   │               │
│           │   │  ┌─────────┐  ┌─────────┐          │   │               │
│           │   │  │Widget 3 │  │Widget 4 │          │   │               │
│           │   │  └─────────┘  └─────────┘          │   │               │
│           │   └─────────────────────────────────────┘   │               │

│                           FOOTER (fixed bottom)                         │
└─────────────────────────────────────────────────────────────────────────┘
Ключевые характеристики:

Страница никогда не прокручивается (body имеет overflow: hidden)

Каждый блок (навбар, сайдбары, футер, виджеты) имеет свою независимую прокрутку

Виджеты можно перетаскивать, менять размеры, добавлять/удалять динамически

Границы виджетов можно двигать (resize)

    <script>
        // Инициализация GridStack
        let grid = null;
        
        // Функция для создания случайного ID
        function generateId() {
            return 'widget_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
        }
        
        // Создание виджета по типу
        function createWidget(type, id = null, title = null, x = 0, y = 0, w = 4, h = 3) {
            const widgetId = id || generateId();
            const widgetTitle = title || getWidgetTitle(type);
            
            // Создаём элемент виджета
            const widget = document.createElement('div');
            widget.className = 'grid-stack-item';
            widget.setAttribute('gs-id', widgetId);
            widget.setAttribute('gs-x', x);
            widget.setAttribute('gs-y', y);
            widget.setAttribute('gs-w', w);
            widget.setAttribute('gs-h', h);
            
            widget.innerHTML = `
                <div class="grid-stack-item-content">
                    <div class="widget-header">
                        <span class="widget-title">${widgetTitle}</span>
                        <div class="widget-actions">
                            <button class="widget-refresh" title="Обновить">🔄</button>
                            <button class="widget-remove" title="Удалить">✖</button>
                        </div>
                    </div>
                    <div class="widget-body" id="${widgetId}_body">
                        <div style="height: 100%;" id="${widgetId}_content">Загрузка...</div>
                    </div>
                </div>
            `;
            
            grid.addWidget(widget);
            
            // Заполняем содержимое
            renderWidgetContent(widgetId, type);
            
            // Обработчики кнопок
            const removeBtn = widget.querySelector('.widget-remove');
            removeBtn.addEventListener('click', () => grid.removeWidget(widget));
            
            const refreshBtn = widget.querySelector('.widget-refresh');
            refreshBtn.addEventListener('click', () => renderWidgetContent(widgetId, type));
            
            return widgetId;
        }
        
        // Получить заголовок виджета
        function getWidgetTitle(type) {
            const titles = {
                'table': '📋 Таблица данных',
                'chart': '📊 График',
                'stats': '📈 Статистика',
                'text': '📝 Заметки',
                'calendar': '📅 Календарь'
            };
            return titles[type] || 'Виджет';
        }
        
        // Рендер содержимого виджета
        async function renderWidgetContent(widgetId, type) {
            const container = document.getElementById(`${widgetId}_content`);
            if (!container) return;
            
            if (type === 'table') {
                // Пример: AG Grid таблица
                container.innerHTML = `<div id="${widgetId}_grid" style="height: 100%; width: 100%;"></div>`;
                const gridDiv = document.getElementById(`${widgetId}_grid`);
                
                // Демо-данные
                const rowData = [
                    { id: 1, name: 'Платеж #1', amount: 1500, status: 'Успешно' },
                    { id: 2, name: 'Платеж #2', amount: 2500, status: 'Успешно' },
                    { id: 3, name: 'Платеж #3', amount: 500, status: 'Ошибка' },
                ];
                
                new agGrid.Grid(gridDiv, {
                    columnDefs: [
                        { field: 'id', headerName: 'ID', width: 80 },
                        { field: 'name', headerName: 'Название', editable: true },
                        { field: 'amount', headerName: 'Сумма', editable: true },
                        { field: 'status', headerName: 'Статус' }
                    ],
                    rowData: rowData,
                    pagination: true,
                    paginationPageSize: 10,
                    domLayout: 'autoHeight'
                });
                
            } else if (type === 'chart') {
                // ECharts график
                container.innerHTML = `<div id="${widgetId}_chart" style="height: 100%; width: 100%;"></div>`;
                const chartDiv = document.getElementById(`${widgetId}_chart`);
                const chart = echarts.init(chartDiv);
                chart.setOption({
                    title: { text: 'Динамика платежей' },
                    tooltip: { trigger: 'axis' },
                    xAxis: { type: 'category', data: ['Янв', 'Фев', 'Мар', 'Апр', 'Май'] },
                    yAxis: { type: 'value' },
                    series: [{ type: 'bar', data: [120, 200, 150, 80, 70] }]
                });
                
            } else if (type === 'stats') {
                container.innerHTML = `
                    <div style="display: grid; grid-template-columns: 1fr 1fr; gap: 10px;">
                        <div style="background: #e8f5e9; padding: 15px; border-radius: 8px; text-align: center;">
                            <div style="font-size: 24px; font-weight: bold;">1,234</div>
                            <div style="font-size: 12px;">Всего платежей</div>
                        </div>
                        <div style="background: #e3f2fd; padding: 15px; border-radius: 8px; text-align: center;">
                            <div style="font-size: 24px; font-weight: bold;">₽ 45,678</div>
                            <div style="font-size: 12px;">Общая сумма</div>
                        </div>
                        <div style="background: #fff3e0; padding: 15px; border-radius: 8px; text-align: center;">
                            <div style="font-size: 24px; font-weight: bold;">89%</div>
                            <div style="font-size: 12px;">Успешных</div>
                        </div>
                        <div style="background: #fce4ec; padding: 15px; border-radius: 8px; text-align: center;">
                            <div style="font-size: 24px; font-weight: bold;">+12%</div>
                            <div style="font-size: 12px;">Рост</div>
                        </div>
                    </div>
                `;
                
            } else if (type === 'text') {
                container.innerHTML = `
                    <textarea style="width: 100%; height: 100%; border: none; resize: none; font-family: inherit; padding: 10px;" placeholder="Введите текст заметки...">Здесь можно писать заметки, которые сохранятся в конфигурации дашборда.</textarea>
                `;
                
            } else if (type === 'calendar') {
                container.innerHTML = `
                    <div style="text-align: center; padding: 20px;">
                        <div style="font-size: 32px;">📅</div>
                        <div>Календарь событий</div>
                        <div style="margin-top: 10px; font-size: 12px; color: #666;">Здесь будет календарь</div>
                    </div>
                `;
            }
        }
        
        // Сохранение конфигурации дашборда
        function saveDashboard() {
            const widgets = [];
            const items = grid.getGridItems();
            
            items.forEach(item => {
                const id = item.getAttribute('gs-id');
                const type = item.querySelector('.widget-title')?.innerText.includes('Таблица') ? 'table' :
                             item.querySelector('.widget-title')?.innerText.includes('График') ? 'chart' : 'stats';
                
                widgets.push({
                    id: id,
                    type: type,
                    title: item.querySelector('.widget-title')?.innerText,
                    x: parseInt(item.getAttribute('gs-x')),
                    y: parseInt(item.getAttribute('gs-y')),
                    w: parseInt(item.getAttribute('gs-w')),
                    h: parseInt(item.getAttribute('gs-h'))
                });
            });
            
            const config = {
                layout: 'gridstack',
                widgets: widgets,
                savedAt: new Date().toISOString()
            };
            
            localStorage.setItem('dashboard_config', JSON.stringify(config));
            alert('Дашборд сохранён');
        }
        
        // Загрузка конфигурации дашборда
        function loadDashboard() {
            const saved = localStorage.getItem('dashboard_config');
            if (!saved) return;
            
            const config = JSON.parse(saved);
            grid.removeAllWidgets();
            
            config.widgets.forEach(w => {
                createWidget(w.type, w.id, w.title, w.x, w.y, w.w, w.h);
            });
        }
        
        // Сброс дашборда (виджеты по умолчанию)
        function resetDashboard() {
            localStorage.removeItem('dashboard_config');
            grid.removeAllWidgets();
            
            // Виджеты по умолчанию
            createWidget('table', null, 'Платежи', 0, 0, 6, 4);
            createWidget('chart', null, 'График платежей', 6, 0, 6, 4);
            createWidget('stats', null, 'Статистика', 0, 4, 6, 3);
            createWidget('text', null, 'Заметки', 6, 4, 6, 3);
        }
        
        // Инициализация drag & drop из правого сайдбара
        function initDragAndDrop() {
            const widgetsList = document.querySelectorAll('.widget-item');
            
            widgetsList.forEach(widget => {
                widget.addEventListener('dragstart', (e) => {
                    e.dataTransfer.setData('text/plain', widget.getAttribute('data-type'));
                    e.dataTransfer.effectAllowed = 'copy';
                });
            });
            
            const gridContainer = document.querySelector('.grid-stack');
            gridContainer.addEventListener('dragover', (e) => {
                e.preventDefault();
                e.dataTransfer.dropEffect = 'copy';
            });
            
            gridContainer.addEventListener('drop', (e) => {
                e.preventDefault();
                const type = e.dataTransfer.getData('text/plain');
                if (type) {
                    // Находим координаты в сетке
                    const rect = gridContainer.getBoundingClientRect();
                    const x = Math.floor((e.clientX - rect.left) / 200); // примерная ширина колонки
                    const y = grid.getRow() + 1;
                    createWidget(type, null, null, Math.max(0, x), y, 4, 3);
                }
            });
        }
        
        // Инициализация при загрузке
        document.addEventListener('DOMContentLoaded', () => {
            // Инициализируем GridStack
            grid = GridStack.init({
                column: 12,
                cellHeight: 100,
                verticalMargin: 10,
                minRow: 2,
                resizable: {
                    handles: 'e, se, s, sw, w'
                },
                draggable: {
                    handle: '.widget-header'
                }
            });
            
            // Загружаем сохранённую конфигурацию или виджеты по умолчанию
            const savedConfig = localStorage.getItem('dashboard_config');
            if (savedConfig) {
                loadDashboard();
            } else {
                resetDashboard();
            }
            
            // Инициализируем drag & drop
            initDragAndDrop();
            
            // Кнопки управления
            document.getElementById('save-dashboard').addEventListener('click', saveDashboard);
            document.getElementById('reset-dashboard').addEventListener('click', resetDashboard);
        });
    </script>

Что такое GridStack
GridStack.js — это библиотека для создания интерактивных сеток (dashboard), где виджеты можно:

Перетаскивать (drag & drop)

Менять размер (resize)

Добавлять/удалять динамически

Сетка адаптируется под количество колонок

Это фундамент, на котором строятся современные дашборды. Именно GridStack позволяет реализовать ту гибкость, которую вы хотите: независимые виджеты, которые пользователь сам расставляет.

GridStack не даёт готового дизайна — он даёт механику. Дизайн вы делаете сами (или берёте готовые стили).

# dashboard/models.py
from django.db import models
from django.contrib.auth import get_user_model

User = get_user_model()

class DashboardPage(models.Model):
    """Страница дашборда (пользователь может создавать несколько страниц)"""
    user = models.ForeignKey(User, on_delete=models.CASCADE, related_name='dashboard_pages')
    name = models.CharField(max_length=100, verbose_name='Название страницы')
    slug = models.SlugField(max_length=100, verbose_name='URL')
    parent = models.ForeignKey('self', on_delete=models.CASCADE, null=True, blank=True, related_name='children')
    order = models.IntegerField(default=0, verbose_name='Порядок')
    is_default = models.BooleanField(default=False, verbose_name='Главная страница')
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        unique_together = ['user', 'slug']
        ordering = ['parent__id', 'order', 'name']
        verbose_name = 'Страница дашборда'
        verbose_name_plural = 'Страницы дашборда'

    def __str__(self):
        return f"{self.user.username} / {self.name}"

class DashboardWidget(models.Model):
    """Виджет на странице дашборда"""
    WIDGET_TYPES = [
        ('text', 'Текстовый блок'),
        ('table', 'Таблица'),
        ('chart', 'График'),
        ('stats', 'Статистика'),
        ('calendar', 'Календарь'),
        ('iframe', 'Внешняя страница'),
        ('custom', 'Пользовательский'),
    ]

    page = models.ForeignKey(DashboardPage, on_delete=models.CASCADE, related_name='widgets')
    widget_type = models.CharField(max_length=20, choices=WIDGET_TYPES, default='text')
    title = models.CharField(max_length=200, blank=True, verbose_name='Заголовок')
    content = models.JSONField(default=dict, verbose_name='Содержимое')
    position_x = models.IntegerField(default=0, verbose_name='X позиция (колонка)')
    position_y = models.IntegerField(default=0, verbose_name='Y позиция (ряд)')
    width = models.IntegerField(default=4, verbose_name='Ширина (колонки)')
    height = models.IntegerField(default=3, verbose_name='Высота (ячейки)')
    order = models.IntegerField(default=0, verbose_name='Порядок')
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['page', 'order', 'position_y', 'position_x']
        verbose_name = 'Виджет'
        verbose_name_plural = 'Виджеты'

    def __str__(self):
        return f"{self.page.name} / {self.get_widget_type_display()} - {self.title or 'без названия'}"

📡 API для работы с дашбордом (Django REST Framework)
python
# dashboard/api.py
from rest_framework import viewsets, serializers, permissions
from rest_framework.decorators import action
from rest_framework.response import Response
from .models import DashboardPage, DashboardWidget

class DashboardPageSerializer(serializers.ModelSerializer):
    class Meta:
        model = DashboardPage
        fields = ['id', 'name', 'slug', 'parent', 'order', 'is_default']

class DashboardWidgetSerializer(serializers.ModelSerializer):
    class Meta:
        model = DashboardWidget
        fields = ['id', 'widget_type', 'title', 'content', 'position_x', 'position_y', 'width', 'height', 'order']

class DashboardPageViewSet(viewsets.ModelViewSet):
    serializer_class = DashboardPageSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        return DashboardPage.objects.filter(user=self.request.user)

    @action(detail=True, methods=['get', 'post'])
    def widgets(self, request, pk=None):
        page = self.get_object()
        if request.method == 'GET':
            widgets = page.widgets.all()
            serializer = DashboardWidgetSerializer(widgets, many=True)
            return Response(serializer.data)
        elif request.method == 'POST':
            serializer = DashboardWidgetSerializer(data=request.data)
            if serializer.is_valid():
                serializer.save(page=page)
                return Response(serializer.data)
            return Response(serializer.errors, status=400)

    @action(detail=False, methods=['post'])
    def reorder(self, request):
        """Обновление всех виджетов страницы (позиции, размеры)"""
        page_id = request.data.get('page_id')
        widgets_data = request.data.get('widgets', [])
        page = DashboardPage.objects.get(id=page_id, user=request.user)

        for w_data in widgets_data:
            widget = DashboardWidget.objects.get(id=w_data['id'], page=page)
            widget.position_x = w_data.get('x', widget.position_x)
            widget.position_y = w_data.get('y', widget.position_y)
            widget.width = w_data.get('w', widget.width)
            widget.height = w_data.get('h', widget.height)
            widget.save()

        return Response({'status': 'ok'})

<script>
    let grid = null;
    let currentPageId = null;
    let quillEditors = {};  // хранилище редакторов

    // Получение CSRF токена
    function getCookie(name) {
        let cookieValue = null;
        if (document.cookie && document.cookie !== '') {
            const cookies = document.cookie.split(';');
            for (let i = 0; i < cookies.length; i++) {
                const cookie = cookies[i].trim();
                if (cookie.substring(0, name.length + 1) === (name + '=')) {
                    cookieValue = decodeURIComponent(cookie.substring(name.length + 1));
                    break;
                }
            }
        }
        return cookieValue;
    }

    // Загрузка страниц
    async function loadPages() {
        const res = await fetch('/api/pages/');
        const pages = await res.json();
        const container = document.getElementById('page-tree');
        container.innerHTML = buildTree(pages, null);
        if (pages.length > 0 && !currentPageId) {
            const defaultPage = pages.find(p => p.is_default) || pages[0];
            await loadPage(defaultPage.id);
        }
    }

    function buildTree(pages, parentId) {
        const children = pages.filter(p => p.parent === parentId);
        if (children.length === 0) return '';
        let html = '<ul class="page-tree">';
        children.forEach(p => {
            html += `
                <li data-id="${p.id}" class="${currentPageId === p.id ? 'active' : ''}">
                    <span>${escapeHtml(p.name)}</span>
                    <span class="page-actions">
                        <button onclick="editPage(${p.id})">✏️</button>
                        <button onclick="deletePage(${p.id})">🗑️</button>
                    </span>
                </li>
            `;
            html += buildTree(pages, p.id);
        });
        html += '</ul>';
        return html;
    }

    function escapeHtml(str) {
        if (!str) return '';
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        });
    }

    // Загрузка страницы
    async function loadPage(pageId) {
        currentPageId = pageId;
        await loadPages(); // обновляем активную подсветку

        const res = await fetch(`/api/pages/${pageId}/widgets/`);
        const widgets = await res.json();

        grid.removeAllWidgets();

        for (const w of widgets) {
            createWidgetFromData(w);
        }
    }

    // Создание виджета из данных
    function createWidgetFromData(w) {
        const widgetId = `widget_${w.id}`;
        const node = {
            id: widgetId,
            x: w.position_x,
            y: w.position_y,
            w: w.width,
            h: w.height,
            content: w.content,
            widgetType: w.widget_type,
            title: w.title
        };
        addWidgetToGrid(node);
    }

    // Добавление виджета в сетку
    function addWidgetToGrid(node) {
        const widget = document.createElement('div');
        widget.className = 'grid-stack-item';
        widget.setAttribute('gs-id', node.id);
        widget.setAttribute('gs-x', node.x);
        widget.setAttribute('gs-y', node.y);
        widget.setAttribute('gs-w', node.w);
        widget.setAttribute('gs-h', node.h);

        widget.innerHTML = `
            <div class="grid-stack-item-content">
                <div class="widget-header">
                    <span class="widget-title">${escapeHtml(node.title || getWidgetTitle(node.widgetType))}</span>
                    <div class="widget-actions">
                        <button class="widget-edit" title="Редактировать">✏️</button>
                        <button class="widget-refresh" title="Обновить">🔄</button>
                        <button class="widget-remove" title="Удалить">✖</button>
                    </div>
                </div>
                <div class="widget-body" id="${node.id}_body"></div>
            </div>
        `;

        grid.addWidget(widget);
        renderWidgetContent(node.id, node.widgetType, node.content);

        // Обработчики
        widget.querySelector('.widget-remove').addEventListener('click', () => removeWidget(node.id));
        widget.querySelector('.widget-edit').addEventListener('click', () => editWidgetContent(node.id, node.widgetType));
        widget.querySelector('.widget-refresh').addEventListener('click', () => refreshWidget(node.id, node.widgetType));
    }

    function getWidgetTitle(type) {
        const titles = { 'text': 'Текстовый блок', 'table': 'Таблица', 'stats': 'Статистика', 'calendar': 'Календарь' };
        return titles[type] || 'Виджет';
    }

    function renderWidgetContent(widgetId, type, content) {
        const container = document.getElementById(`${widgetId}_body`);
        if (!container) return;

        if (type === 'text') {
            const editorId = `editor_${widgetId}`;
            container.innerHTML = `<div id="${editorId}" style="height: 100%;"></div>`;
            const quill = new Quill(`#${editorId}`, {
                theme: 'snow',
                placeholder: 'Введите текст...',
                modules: { toolbar: [
                    ['bold', 'italic', 'underline', 'strike'],
                    ['blockquote', 'code-block'],
                    [{ 'header': 1 }, { 'header': 2 }],
                    [{ 'list': 'ordered'}, { 'list': 'bullet' }],
                    ['link', 'image']
                ] }
            });
            if (content && content.html) quill.root.innerHTML = content.html;
            quillEditors[widgetId] = quill;
        } else if (type === 'table') {
            container.innerHTML = `<div id="${widgetId}_grid" style="height: 100%; width: 100%;"></div>`;
            const gridDiv = document.getElementById(`${widgetId}_grid`);
            new agGrid.Grid(gridDiv, {
                columnDefs: [{ field: 'name', headerName: 'Название' }, { field: 'value', headerName: 'Значение' }],
                rowData: content?.data || [{ name: 'Пример', value: 100 }],
                pagination: true,
                paginationPageSize: 10
            });
        } else if (type === 'stats') {
            container.innerHTML = `
                <div style="display: grid; grid-template-columns: 1fr 1fr; gap: 10px;">
                    <div style="background: #e8f5e9; padding: 15px; border-radius: 8px; text-align: center;">
                        <div style="font-size: 24px; font-weight: bold;">${content?.stat1 || '123'}</div>
                        <div>Всего</div>
                    </div>
                    <div style="background: #e3f2fd; padding: 15px; border-radius: 8px; text-align: center;">
                        <div style="font-size: 24px; font-weight: bold;">${content?.stat2 || '45'}</div>
                        <div>Активных</div>
                    </div>
                </div>
            `;
        } else if (type === 'calendar') {
            container.innerHTML = `<div style="text-align: center;">📅 Календарь (демо)</div>`;
        }
    }

    async function removeWidget(gridId) {
        const id = gridId.replace('widget_', '');
        await fetch(`/api/widgets/${id}/`, { method: 'DELETE', headers: { 'X-CSRFToken': getCookie('csrftoken') } });
        const item = document.querySelector(`.grid-stack-item[gs-id="${gridId}"]`);
        if (item) grid.removeWidget(item);
    }

    async function editWidgetContent(gridId, type) {
        const id = gridId.replace('widget_', '');
        const newTitle = prompt('Введите заголовок виджета:', document.querySelector(`.grid-stack-item[gs-id="${gridId}"] .widget-title`).innerText);
        if (newTitle) {
            document.querySelector(`.grid-stack-item[gs-id="${gridId}"] .widget-title`).innerText = newTitle;
            await fetch(`/api/widgets/${id}/`, {
                method: 'PATCH',
                headers: { 'Content-Type': 'application/json', 'X-CSRFToken': getCookie('csrftoken') },
                body: JSON.stringify({ title: newTitle })
            });
        }
    }

    async function refreshWidget(gridId, type) {
        const id = gridId.replace('widget_', '');
        const res = await fetch(`/api/widgets/${id}/`);
        const data = await res.json();
        renderWidgetContent(gridId, type, data.content);
    }

    async function saveDashboard() {
        const widgets = [];
        const items = grid.getGridItems();
        for (const item of items) {
            const gridId = item.getAttribute('gs-id');
            const id = gridId.replace('widget_', '');
            const content = quillEditors[gridId] ? { html: quillEditors[gridId].root.innerHTML } : {};
            await fetch(`/api/widgets/${id}/`, {
                method: 'PATCH',
                headers: { 'Content-Type': 'application/json', 'X-CSRFToken': getCookie('csrftoken') },
                body: JSON.stringify({
                    position_x: parseInt(item.getAttribute('gs-x')),
                    position_y: parseInt(item.getAttribute('gs-y')),
                    width: parseInt(item.getAttribute('gs-w')),
                    height: parseInt(item.getAttribute('gs-h')),
                    content: content
                })
            });
        }
        alert('Дашборд сохранён');
    }

                });
                const newWidget = await res.json();
                createWidgetFromData(newWidget);
            }
        });
        document.querySelector('.grid-stack').addEventListener('dragover', e => e.preventDefault());
    });

urlpatterns = [
    path('', views.dashboard, name='dashboard'),
