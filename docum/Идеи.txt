    parent = models.ForeignKey(

        null=True,
        blank=True,
        related_name='children',
        verbose_name='Родительская страница'
    )
    title = models.CharField(

        verbose_name='Название страницы'
    )
    slug = models.SlugField(
        max_length=200,
        verbose_name='URL (уникальный в пределах сайта)'
    )
    site_id = models.IntegerField(
        default=0,
        verbose_name='ID сайта (для группировки страниц в сайты)'
    )
    site_name = models.CharField(
        max_length=100,
        blank=True,
        verbose_name='Название сайта'
    )
    icon = models.CharField(
        max_length=50,
        blank=True,
        verbose_name='Иконка (Font Awesome класс)'
    )
    is_default = models.BooleanField(
        default=False,
        verbose_name='Стартовая страница сайта'
    )
    is_visible = models.BooleanField(
        default=True,
        verbose_name='Отображать в меню'
    )
    order = models.IntegerField(
        default=0,
        verbose_name='Порядок сортировки'
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        verbose_name = 'Страница дашборда'
        verbose_name_plural = 'Страницы дашборда'
        ordering = ['site_id', 'parent__id', 'order', 'title']
        unique_together = [['user', 'site_id', 'slug']]
        indexes = [
            models.Index(fields=['user', 'site_id']),
            models.Index(fields=['user', 'parent']),
        ]

    def __str__(self):
        prefix = f"[{self.site_name}] " if self.site_name else ""
        return f"{prefix}{self.title}"

    def get_full_path(self):
        """Возвращает полный путь к странице (для навигации)"""
        if self.parent:
            return f"{self.parent.get_full_path()}/{self.slug}"
        return f"/{self.slug}"

    def get_site_pages(self):
        """Возвращает все страницы текущего сайта"""
        return DashboardPage.objects.filter(
            user=self.user,
            site_id=self.site_id
        ).select_related('parent')

    def get_tree(self):
        """Возвращает дерево страниц текущего сайта"""
        pages = self.get_site_pages()
        return self._build_tree(pages, None)

    @staticmethod
    def _build_tree(pages, parent_id):
        tree = []
        for page in pages.filter(parent_id=parent_id):
            tree.append({
                'id': page.id,
                'title': page.title,
                'slug': page.slug,
                'icon': page.icon,
                'is_default': page.is_default,
                'order': page.order,
                'children': DashboardPage._build_tree(pages, page.id)
            })
        return sorted(tree, key=lambda x: (x['order'], x['title']))

    full_path = serializers.SerializerMethodField()

        model = DashboardPage
        fields = ['id', 'parent', 'title', 'slug', 'site_id', 'site_name', 
                  'icon', 'is_default', 'is_visible', 'order', 'children', 
                  'full_path', 'created_at', 'updated_at']

    def get_children(self, obj):
        return DashboardPageSerializer(obj.children.all(), many=True).data

    def get_full_path(self, obj):
        return obj.get_full_path()


class DashboardPageViewSet(viewsets.ModelViewSet):
    serializer_class = DashboardPageSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        return DashboardPage.objects.filter(user=self.request.user)

    @action(detail=False, methods=['get'])
    def sites(self, request):
        """Возвращает список сайтов пользователя с группировкой страниц"""
        pages = DashboardPage.objects.filter(user=request.user)
        sites = pages.values('site_id', 'site_name').distinct().order_by('site_id')
        result = []
        for site in sites:
            site_pages = pages.filter(site_id=site['site_id'])
            result.append({
                'site_id': site['site_id'],
                'site_name': site['site_name'] or f'Сайт #{site["site_id"]}',
                'pages': DashboardPageSerializer(site_pages.filter(parent=None), many=True).data
            })
        return Response(result)

    @action(detail=False, methods=['post'])
    def create_site(self, request):
        """Создаёт новый сайт (группу страниц)"""
        site_id = DashboardPage.objects.filter(user=request.user).aggregate(
            max_id=models.Max('site_id')
        )['max_id'] or 0
        site_id += 1
        return Response({'site_id': site_id})

    @action(detail=False, methods=['post'])
    def move_page(self, request):
        """Перемещает страницу в новую позицию (сортировка, смена родителя)"""
        page_id = request.data.get('page_id')
        new_parent_id = request.data.get('new_parent_id')
        new_order = request.data.get('new_order', 0)

        page = DashboardPage.objects.get(id=page_id, user=request.user)
        if new_parent_id:
            page.parent_id = new_parent_id
        page.order = new_order
        page.save()

        # Перенумеровываем остальные страницы того же родителя
        siblings = DashboardPage.objects.filter(
            user=request.user,
            parent_id=page.parent_id
        ).exclude(id=page_id).order_by('order')
        for i, sibling in enumerate(siblings):
            sibling.order = i
            sibling.save()

        return Response({'status': 'ok'})

        let cookieValue = null;
        if (document.cookie && document.cookie !== '') {

                const cookie = cookies[i].trim();
                if (cookie.substring(0, name.length + 1) === (name + '=')) {

# models.py
from django.db import models
from django.contrib.auth import get_user_model

User = get_user_model()

class UserSubscription(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    trial_start = models.DateTimeField(auto_now_add=True)
    trial_end = models.DateTimeField(null=True, blank=True)
    plan = models.CharField(max_length=20, default='trial')  # trial, pro, business
    is_active = models.BooleanField(default=True)

    def is_trial_valid(self):
        if not self.trial_end:
            # 30 дней от регистрации
            self.trial_end = self.trial_start + timedelta(days=30)
            self.save()
        return timezone.now() < self.trial_end and self.plan == 'trial'


# middleware.py
class TrialMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        if request.user.is_authenticated:
            subscription = UserSubscription.objects.get(user=request.user)
            if subscription.plan == 'trial' and not subscription.is_trial_valid():
                # Пробный период закончился
                request.trial_expired = True
                return redirect('subscription_required')
        return self.get_response(request)
